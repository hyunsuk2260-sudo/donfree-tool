<script>
// 구글 뉴스 실시간 RSS 자동 가져오기 스크립트
document.addEventListener("DOMContentLoaded", function() {
    // 대한민국 실시간 주요 뉴스 피드 URL
    const rssUrl = 'https://news.google.com/rss?hl=ko&gl=KR&ceid=KR:ko';
    // RSS를 자바스크립트가 읽을 수 있게 바꿔주는 무료 API
    const apiUrl = 'https://api.rss2json.com/v1/api.json?rss_url=' + encodeURIComponent(rssUrl);

    fetch(apiUrl)
        .then(response => response.json())
        .then(data => {
            const list = document.getElementById('auto-news-list');
            if(!list) return;
            list.innerHTML = ''; 
            
            // 최신 뉴스 딱 6개만 깔끔하게 노출
            const articles = data.items.slice(0, 6);
            
            articles.forEach(article => {
                // 제목 뒤에 붙은 불필요한 언론사 이름 잘라내서 깔끔하게 만들기
                let cleanTitle = article.title.split(' - ')[0];
                
                const li = document.createElement('li');
                li.className = 'pt-3 first:pt-0';
                li.innerHTML = `
                    <a href="${article.link}" target="_blank" class="group flex items-center justify-between gap-3 text-decoration-none">
                        <span class="text-sm sm:text-base text-gray-800 group-hover:text-blue-600 transition-colors line-clamp-1 font-medium">${cleanTitle}</span>
                        <span class="text-[10px] text-gray-400 whitespace-nowrap bg-gray-100 px-2 py-1 rounded">새창이동 ↗</span>
                    </a>
                `;
                list.appendChild(li);
            });
        })
        .catch(error => {
            const list = document.getElementById('auto-news-list');
            if(list) list.innerHTML = '<li class="text-sm text-red-500 py-2 text-center">뉴스를 불러오는데 실패했습니다. 새로고침 해주세요.</li>';
        });
});
</script>
